-- staffing_who_works.sql (Nov 7 2026): step 1b of the staff schedule builder ("who works") -- staff_profiles.aim_40: managers marked to aim for 40 hours (others 30–40) -- bd_floater_pct(): the weekend floater's labour check. The builder sends the day's planned shifts (paid minutes per -- person) and how many extra walk-in people it expects; this returns the day's labour % only (wages stay on the server), -- from that day's booked revenue plus the expected walk-ins at the day's average price per person. ALTER TABLE public.staff_profiles ADD COLUMN IF NOT EXISTS aim_40 boolean NOT NULL DEFAULT false; COMMENT ON COLUMN public.staff_profiles.aim_40 IS 'Managers: aim for 40 hours a week (others are scheduled 30–40)'; CREATE OR REPLACE FUNCTION public.bd_floater_pct(p_studio_id text, p_day date, p_extra_people numeric, p_shifts jsonb) RETURNS integer LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path TO 'public' AS $function$ DECLARE v_rev numeric := 0; v_people numeric := 0; v_avg numeric; v_studio_wage numeric; v_lab numeric := 0; x jsonb; BEGIN IF coalesce(nullif(current_setting('request.jwt.claims', true), '')::jsonb ->> 'role', '') NOT IN ('', 'service_role') AND NOT public.is_home_office() THEN RAISE EXCEPTION 'Not authorized' USING ERRCODE = '42501'; END IF; -- the day's booked revenue (same as the Hub's labour %), and the people booked into that day's sessions SELECT coalesce(sum(revenue), 0) INTO v_rev FROM public.hub_day_revenue(p_studio_id, p_day, p_day); SELECT coalesce(sum(b.qty), 0) INTO v_people FROM bookings b JOIN sessions s ON s.id = b.session_id WHERE s.studio_id = p_studio_id AND s.session_date = p_day AND b.qty > 0 AND b.returned_at IS NULL; -- the average price per person that day (kept between $20 and $80); $35 when nothing's booked yet v_avg := CASE WHEN v_people > 0 AND v_rev > 0 THEN least(80, greatest(20, v_rev / v_people)) ELSE 35 END; -- planned wages: each person's hourly wage; an open shift (no one yet) at the studio's average SELECT avg(hourly_wage) INTO v_studio_wage FROM staff_profiles WHERE is_active IS NOT FALSE AND hourly_wage IS NOT NULL AND (home_studio_id = p_studio_id OR p_studio_id = ANY (studio_ids)); FOR x IN SELECT * FROM jsonb_array_elements(coalesce(p_shifts, '[]'::jsonb)) LOOP v_lab := v_lab + coalesce((x->>'mins')::numeric, 0) / 60.0 * coalesce((SELECT hourly_wage FROM staff_profiles WHERE id = NULLIF(x->>'staff_id', '')::uuid), v_studio_wage, 18); END LOOP; IF v_rev + greatest(coalesce(p_extra_people, 0), 0) * v_avg <= 0 THEN RETURN NULL; END IF; RETURN round(v_lab / (v_rev + greatest(coalesce(p_extra_people, 0), 0) * v_avg) * 100); END $function$; REVOKE ALL ON FUNCTION public.bd_floater_pct(text, date, numeric, jsonb) FROM PUBLIC, anon; GRANT EXECUTE ON FUNCTION public.bd_floater_pct(text, date, numeric, jsonb) TO authenticated; NOTIFY pgrst, 'reload schema'; -- Mark the managers who aim for 40 hours (edit the names), then check: -- UPDATE staff_profiles SET aim_40 = true WHERE role = 'manager' AND display_name IN ('First Last', 'First Last'); SELECT display_name, home_studio_id, aim_40, min_weekly_hours, max_weekly_hours FROM staff_profiles WHERE role = 'manager' AND is_active IS NOT FALSE ORDER BY home_studio_id, display_name;
